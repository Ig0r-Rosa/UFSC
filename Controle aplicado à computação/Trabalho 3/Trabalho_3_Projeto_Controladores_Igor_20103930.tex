\section*{Questão 1}
%==============================================================================

\subsection*{Enunciado}

Planta $G(s)=\dfrac{1}{(s+2)(s+5)}$ em realimentação unitária. Requisitos:
erro nulo ao degrau; resposta \textbf{sobreamortecida} com $T_s=1\,\mathrm{s}$
(critério de 5\%).

\subsection*{Passo a passo do projeto}

\begin{center}
\begin{tabular}{@{}ll@{}}
  \toprule
  Requisito & Implicação \\
  \midrule
  Erro nulo ao degrau & Integrador em $C(s)$ $\Rightarrow$ malha aberta \textbf{tipo~1} \\
  Sobreamortecida & Polos em MF \textbf{reais} (sem oscilação) \\
  $T_s=1\,\mathrm{s}$ (5\%) & $T_s\approx 4/|\sigma|$ $\Rightarrow$ polo dominante em $s\approx -4$ \\
  \bottomrule
\end{tabular}
\end{center}

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item Interpretar requisitos: erro nulo $\Rightarrow$ integrador; resposta
        sobreamortecida $\Rightarrow$ polos reais; $T_s=1\,\mathrm{s}$
        $\Rightarrow$ $\sigma\approx 4\,\mathrm{rad/s}$.
  \item Escolher \textbf{PID} ($C(s)=\frac{K_d s^2+K_p s+K_i}{s}$): três
        parâmetros para posicionar três polos em malha fechada.
  \item Obter a equação característica com realimentação unitária.
  \item Colocar \textbf{três polos reais iguais} em $s=-4$.
  \item Expandir o polinômio desejado e comparar coeficientes.
\end{enumerate}

Com PID $C(s)=K_p+K_i/s+K_d s$, a equação característica é
\[
  s^3+(7+K_d)s^2+(10+K_p)s+K_i=0.
\]
Escolhem-se \textbf{três polos reais iguais} em $s=-4$:
\[
  (s+4)^3=s^3+12s^2+48s+64.
\]
Comparando coeficientes: $K_d=5$, $K_p=38$, $K_i=64$. Logo,
\[
  C(s)=\frac{5s^2+38s+64}{s}.
\]

  $T_s$ (5\%) & $1{,}0\,\mathrm{s}$ & $\approx 0{,}47\,\mathrm{s}$ & Sim \\
  Polos em MF & Reais em $s\approx -4$ & $s\approx -4$ & Sim \\
  \bottomrule

analítico previu $T_s=4/|\sigma|=1\,\mathrm{s}$ com polos em $s=-4$; a
simulação confirma acomodação \textbf{mais rápida}, ainda dentro da
especificação.

  \bottomrule

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item Requisitos: erro nulo $\Rightarrow$ integrador; $\zeta=0{,}7$

  \item Escolher \textbf{PID} para posicionar par dominante e polo real auxiliar.

$T_s\approx 0{,}3\,\mathrm{s}$ (5\%).

\subsection*{Passo a passo do projeto}

\begin{center}
\begin{tabular}{@{}ll@{}}
  \toprule
  Requisito & Implicação \\
  \midrule
  Erro nulo ao degrau & Planta já é \textbf{tipo~1} ($1/s$) \\
  $\zeta\approx 0{,}5$ & Par complexo dominante \\
  $T_s\approx 0{,}3\,\mathrm{s}$ & $\omega_n=\dfrac{4}{\zeta T_s}=\dfrac{80}{3}\,\mathrm{rad/s}$ \\
  \bottomrule
\end{tabular}
\end{center}

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item A planta já possui integrador ($1/s$) $\Rightarrow$ erro nulo sem
        integrador extra; basta \textbf{PD}.
  \item Calcular $\omega_n=4/(\zeta T_s)=80/3\,\mathrm{rad/s}$ a partir de
        $\zeta=0{,}5$ e $T_s=0{,}3\,\mathrm{s}$.
  \item Definir par dominante em
        $s=-\dfrac{40}{3}\pm j\dfrac{40\sqrt{3}}{3}$.
  \item Obter $K_p$ e $K_d$ pela comparação de coeficientes da equação de
        2ª ordem em malha fechada.
\end{enumerate}

Polos desejados: $s=-\dfrac{40}{3}\pm j\dfrac{40\sqrt{3}}{3}$. Com PD
$C(s)=K_p+K_d s$:
\[
  s^2+(10+10K_d)s+10K_p=0.
\]
Comparando com $s^2+\dfrac{80}{3}s+\dfrac{6400}{9}$:
$K_d=\dfrac{5}{3}$, $K_p=\dfrac{640}{9}$. Logo,
$C(s)=\dfrac{640}{9}+\dfrac{5}{3}s$.

\subsection*{Enunciado}

Planta $G(s)=\dfrac{5}{1+0{,}5s}=\dfrac{10}{s+2}$ em realimentação unitária.
Requisitos: resposta \textbf{sobreamortecida}; $e_{ss}=1\%$ ao degrau;
$T_s\approx 0{,}1\,\mathrm{s}$ (5\%).

\subsection*{Passo a passo do projeto}

\begin{center}
\begin{tabular}{@{}ll@{}}
  \toprule
  Requisito & Implicação \\
  \midrule
  $e_{ss}=1\%$ & Malha tipo~0: $e_{ss}=\dfrac{1}{1+K_p}$, com $K_p=L(0)=99$ \\
  Sobreamortecida & Polos reais e distintos em MF \\
  $T_s\approx 0{,}1\,\mathrm{s}$ & Polo dominante em $s\approx -40\,\mathrm{rad/s}$ \\
  \bottomrule
\end{tabular}
\end{center}

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item Erro de 1\% (não nulo) $\Rightarrow$ malha \textbf{tipo~0}; não se
        usa integrador puro.
  \item Escolher \textbf{compensador de atraso} $C(s)=\frac{as+b}{s+c}$ para
        elevar o ganho DC sem cancelar dinâmica de forma instável.
  \item Impor $L(0)=99$ para $e_{ss}=1\%$.
  \item Colocar polos reais em $s=-40$ e $s=-200$ (sobreamortecido e
        $T_s\approx 4/40=0{,}1\,\mathrm{s}$).
  \item Resolver $a$, $b$, $c$ pela comparação de coeficientes.
\end{enumerate}

Compensador de atraso $C(s)=\dfrac{as+b}{s+c}$. Malha aberta:
$L(s)=\dfrac{5(as+b)}{(s+c)(1+0{,}5s)}$. Equação característica:
\[
  s^2+(2+c+10a)s+2(c+5b)=0.
\]
\textbf{Erro de 1\%:} $L(0)=\dfrac{5b}{c}=99\Rightarrow b=19{,}8\,c$.

\textbf{Polos desejados} $s_1=-40$, $s_2=-200$: $s^2+240s+8000=0$.

Comparando: $c=40$, $b=792$, $a=19{,}8$. Logo,
\[
  C(s)=\frac{19{,}8\,s+792}{s+40}.
\]

\subsection*{Enunciado}

Planta $G(s)=\dfrac{4}{(s+1)(s+3)(s+4)}$ em realimentação unitária.
Requisitos: erro nulo; $\zeta=0{,}7$; $T_s=1{,}5\,\mathrm{s}$ (5\%).

\subsection*{Passo a passo do projeto}

\begin{center}
\begin{tabular}{@{}ll@{}}
  \toprule
  Requisito & Implicação \\
  \midrule
  Erro nulo ao degrau & Integrador em $C(s)$ $\Rightarrow$ malha aberta \textbf{tipo~1} \\
  $\zeta=0{,}7$, $T_s=1{,}5\,\mathrm{s}$ &
  $\omega_n=\dfrac{4}{\zeta T_s}=\dfrac{80}{21}\,\mathrm{rad/s}$ \\
  \bottomrule
\end{tabular}
\end{center}

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item Erro nulo $\Rightarrow$ PID com integrador.
  \item Calcular $\omega_n=80/21\,\mathrm{rad/s}$ a partir de $\zeta=0{,}7$ e
        $T_s=1{,}5\,\mathrm{s}$.
  \item Obter equação característica de 4ª ordem; notar que a soma dos polos
        em MF é fixa em $-8$.
  \item Colocar par dominante e dois polos reais auxiliares ($s=-1$,
        $s=-5/3$) que somam $-8/3$.
  \item Comparar coeficientes e obter $K_d$, $K_p$, $K_i$.
\end{enumerate}

Com PID, a equação característica é
\[
  s^4+8s^3+(19+4K_d)s^2+(12+4K_p)s+4K_i=0.
\]
O coeficiente de $s^3$ é \textbf{fixo em 8} (estrutura da planta), logo a soma
dos polos em MF é $-8$. Com o par dominante em
$s=-\dfrac{8}{3}\pm j\dfrac{8\sqrt{51}}{21}$, os polos auxiliares reais devem
somar $-\dfrac{8}{3}$.

Escolhem-se $s=-1$ e $s=-\dfrac{5}{3}$. Polinômio desejado:
\[
  \left(s^2+\frac{16}{3}s+\frac{6400}{441}\right)(s+1)\left(s+\frac{5}{3}\right)
  =s^4+8s^3+30{,}40s^2+47{,}59s+24{,}19.
\]
Comparando: $K_d=2{,}85$, $K_p=8{,}90$, $K_i=6{,}05$. Logo,
\[
  C(s)=\frac{2{,}85\,s^2+8{,}90\,s+6{,}05}{s}.
\]
O par dominante satisfaz $T_s=4/(\zeta\omega_n)=1{,}5\,\mathrm{s}$.

\begin{tabular}{@{}llll@{}}

\section*{Questão 6}
%==============================================================================

\subsection*{Enunciado}

Mesma planta da questão~5. Projetar controlador \textbf{PI} pelo método de
\textbf{Ziegler-Nichols em malha fechada}.

\subsection*{Passo a passo do projeto}

\textbf{Passos realizados:}
\begin{enumerate}[leftmargin=*]
  \item Fechar a malha com $C(s)=K_p$ e aumentar $K_p$ até surgirem
        \textbf{oscilações sustentadas} (ganho crítico $K_u$).
  \item Medir o período $P_u$ das oscilações.
  \item Aplicar as regras de ZN para PI em malha fechada:
        $K_p=0{,}45\,K_u$, $T_i=P_u/1{,}2$, $K_i=K_p/T_i$.
  \item Montar $C(s)=K_p+K_i/s$ e verificar estabilidade e $e_{ss}$.
\end{enumerate}

\subsection*{Determinação analítica de $K_u$ e $P_u$}

Com $C(s)=K$, a equação característica é
$s^3+8s^2+19s+(12+4K)=0$. Pelo \textbf{critério de Routh-Hurwitz}, oscilações
sustentadas ocorrem quando
\[
  8\cdot 19-(12+4K_u)=0 \quad\Rightarrow\quad K_u=35.
\]
Substituindo $K_u$ e $s=j\omega_u$: da parte imaginária,
$\omega_u^2=19$, logo $\omega_u=\sqrt{19}\,\mathrm{rad/s}$ e
$P_u=2\pi/\omega_u\approx 1{,}44\,\mathrm{s}$.

\subsection*{Controlador PI projetado}

$K_p=0{,}45\times 35=15{,}75$;
$T_i=1{,}44/1{,}2\approx 1{,}20\,\mathrm{s}$;
$K_i=15{,}75/1{,}20\approx 13{,}11$. Logo,
\[
  C(s)=15{,}75+\frac{13{,}11}{s}.
\]
(Regras: $K_p=0{,}45\,K_u$, $T_i=P_u/1{,}2$, $K_i=K_p/T_i$.)

\subsection*{Verificação dos requisitos (simulação)}

\end{tabular}
