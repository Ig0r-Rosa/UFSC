\subsection*{Questão 1 -- Escoamento em tubulação}

\textbf{Informações e instruções:}
\begin{itemize}
    \item Respostas que diferirem do gabarito em até 1,5\,\% serão consideradas corretas, portanto escreva suas respostas com pelo menos 4 algarismos significativos, o que pode ser diferente de um algarismo com 4 casas decimais.
    \item Algumas respostas podem ser consideradas corretas mesmo que difiram do gabarito em mais que 1,5\,\%, se forem corretamente compatíveis com respostas correlatas.
    \item Alguns itens podem ter nota parcial.
    \item A aceleração da gravidade deve ser obrigatoriamente assumida como 9,81~m/s$^2$.
    \item Não deixe campos em branco, nem coloque letras ou sinais nos campos, pois isso ZERA a nota de toda a questão.
    \item Cada nova tentativa tem penalidade de 10\,\% da nota máxima da questão.
\end{itemize}

\textbf{Enunciado:} Um fluido deve ser transportado por 90~m através de uma tubulação com 21~cm de diâmetro e rugosidade equivalente de 0,08~mm. A viscosidade dinâmica, a massa específica e a viscosidade cinemática desse fluido são, respectivamente, 0,01715~N$\cdot$s/m$^2$, 1050~kg/m$^3$ e $16{,}333 \times 10^{-6}$~m$^2$/s.

\begin{enumerate}[label=(\alph*)]
    \item \textbf{(2\%)} A rugosidade relativa do tubo é \resposta{0,000381}.
    \item \textbf{(3\%)} Para que um escoamento permaneça laminar, o número de Reynolds máximo deve ser \resposta{2100}, e com esse número de Reynolds:
    \item \textbf{(57\%)} a velocidade no centro do tubo é \resposta{0,3267}~m/s, a velocidade média é \resposta{0,1633}~m/s, a vazão é \resposta{5,66}~L/s, a variação de pressão entre a entrada e saída da tubulação é \resposta{182,93}~Pa, o fator de atrito é \resposta{0,0305}, o comprimento da região de entrada é \resposta{26,46}~m e a tensão de cisalhamento na parede é \resposta{0,10671}~Pa.
    \item \textbf{(38\%)} No entanto, se a vazão for 961,7~L/s, o número de Reynolds é \resposta{356990}, o fator de atrito calculado com a equação de Colebrook é \resposta{0,0171} e a perda de carga é \resposta{288,4}~m.
\end{enumerate}

\textbf{Distribuição de pontos:}
\begin{itemize}
    \item No item (c): velocidade no centro (7\%), velocidade média (6\%), vazão (5\%), variação de pressão (9\%), fator de atrito (8\%), comprimento da região de entrada (10\%), tensão de cisalhamento (12\%).
    \item No item (d): número de Reynolds (11\%), fator de atrito Colebrook (14\%), perda de carga (13\%).
    \item Respostas com diâmetros utilizados em cm ou convertidos erroneamente reduzem a nota em 20\,\% do valor máximo, assim como vazão em m$^3$/s.
\end{itemize}

}}

\subsection*{Questão 2 -- Diâmetro hidráulico}

\textbf{Enunciado:} Identifique o diâmetro hidráulico dos seguintes dutos. Linhas pretas grossas indicam parede do tubo, e a área ocupada pelo fluido é aquela em azul. Cada nova tentativa tem penalidade de 10\,\% da nota máxima.

\begin{center}
\includegraphics[width=0.55\textwidth]{extracted_images/page3_img3.png}
\end{center}

\textbf{Respostas:}
\begin{enumerate}
    \item Retângulo (largura $A$, altura $B$): \resposta{$\dfrac{2AB}{A+B}$}
    \item Anel (diâmetro externo $A$, diâmetro interno $B$): \resposta{$A - B$}
    \item Semicírculo (raio $A$): \resposta{$2A$}
    \item Semicírculo (raio $A$): \resposta{$\dfrac{2\pi A}{\pi + 2}$}
\end{enumerate}

\subsection*{Questão 3 -- Colisão de jatos}

\textbf{Informações e instruções:}
\begin{itemize}
    \item Respostas que diferirem do gabarito em até 1,0\,\% serão consideradas corretas, portanto escreva suas respostas com pelo menos 4 algarismos significativos.
    \item Algumas respostas podem ser consideradas corretas mesmo que difiram do gabarito em mais que 1,0\,\%, se forem corretamente compatíveis com respostas correlatas.
    \item Alguns itens podem ter nota parcial.
    \item A aceleração da gravidade deve ser obrigatoriamente assumida como 9,81~m/s$^2$.
    \item Não deixe campos em branco, nem coloque letras ou sinais nos campos, pois isso ZERA a nota de toda a questão.
    \item Cada nova tentativa tem penalidade de 10\,\% da nota máxima da questão.
\end{itemize}

\textbf{Enunciado:} Dois jatos de um fluido incompressível colidem e formam um terceiro jato conforme mostrado na figura abaixo.

\begin{center}
\includegraphics[width=0.85\textwidth]{extracted_images/page4_img1.png}
\end{center}

O escoamento ocorre todo na horizontal e os efeitos gravitacionais podem ser desprezados.

Os valores de $D_1$, $D_2$, $V_1$ e $V_2$ são, respectivamente, 13~cm, 9,1~cm, 3,3~m/s e 1,7~m/s.

\begin{enumerate}[label=(\alph*)]
    \item \textbf{(20\%)} As vazões nos locais (1), (2) e (3) são respectivamente, \resposta{0,043802}~m$^3$/s, \resposta{0,011057}~m$^3$/s e \resposta{0,054858}~m$^3$/s.
    \item \textbf{(55\%)} A velocidade, a área e ângulo $\theta$ do jato formado são respectivamente, \resposta{2,66}~m/s, \resposta{0,020646}~m$^2$ e \resposta{82,59}$^\circ$.
    \item \textbf{(25\%)} Mantendo-se os diâmetros dos jatos de entrada inalterados, assim como o ângulo entre esses jatos, a razão entre $V_2$ e $V_1$ deve ser \resposta{1,02} para que o $\theta$ do jato formado seja $63^\circ$.
\end{enumerate}

\textbf{Distribuição de pontos:}
\begin{itemize}
    \item No item (a): vazões nos locais (1), (2) e (3) valem 5\%, 6\% e 9\%, respectivamente.
    \item No item (b): velocidade, área e ângulo $\theta$ valem 20\%, 12\% e 23\%, respectivamente.
    \item Utilizar seno no lugar de cosseno (ou vice-versa) reduz o valor máximo em 25\,\%.
    \item Erro no cálculo da área para vazão reduz o valor máximo em 20\,\%.
\end{itemize}
